```javascript
function getCustomerName(order) {
  // 1. 优先从 order_client 读取（新旧都有）
  if (order.order_client) {
    return order.order_client
  }
  // 2. 如果是新订单但 order_client 为空，从 customer_id 关联查询

  }

}

// 示例：获取商品信息
function getGoodsDisplay(order) {
  // 1. 如果有 order_goods 数组（新订单）
  if (order.order_goods && order.order_goods.length > 0) {
    // 显示第一个商品，如果有多个则显示 "商品名1 等N件"
    const first = order.order_goods[0]
    const count = order.order_goods.length
    if (count > 1) {
      return `${first.goods_name} 等${count}件商品`
    }
    return `${first.goods_name} ${first.spec}`
  }
  // 2. 回退到旧字段
  return order.goods_name || '-'
}

// 示例：获取总重量
function getTotalWeight(order) {
  // 1. 如果有 order_goods 数组
  if (order.order_goods && order.order_goods.length > 0) {
    const total = order.order_goods.reduce((sum, item) => sum + (item.quantity || 0), 0)
    return `${total}kg`
  }
  // 2. 回退到旧字段
  return order.goods_weight || '-'
}

// 示例：获取件数
function getPackageCount(order) {
  // 1. 如果有 order_goods 数组
  if (order.order_goods && order.order_goods.length > 0) {
    const total = order.order_goods.reduce((sum, item) => sum + (item.packages || 0), 0)
    return `${total}件`
  }
  // 2. 回退到旧字段
  return order.goods_quantity || '-'
}
```

```javascript
function matchSearchKeyword(order, keyword) {
  if (!keyword) return true
  
  const searchText = keyword.toLowerCase()
  
  // 客户名称
  if (order.order_client && order.order_client.toLowerCase().includes(searchText)) {
    return true
  }
  
  // 订单编号（新订单）
  if (order.order_number && order.order_number.toLowerCase().includes(searchText)) {
    return true
  }
  
  // 联系人
  if (order.contact_person && order.contact_person.toLowerCase().includes(searchText)) {
    return true
  }
  if (order.receiver_name && order.receiver_name.toLowerCase().includes(searchText)) {
    return true
  }
  
  // 工程项目（新订单）
  if (order.project_name && order.project_name.toLowerCase().includes(searchText)) {
    return true
  }
  
  // 商品名称（旧字段）
  if (order.goods_name && order.goods_name.toLowerCase().includes(searchText)) {
    return true
  }
  
  // 商品明细（新字段）
  if (order.order_goods && order.order_goods.length > 0) {
    const hasMatch = order.order_goods.some(item =>

  }
  
  return false
}
```

### 导出逻辑
```javascript
function exportToExcel(orders) {
  const exportData = []
  
  // 表头
  exportData.push(['序号', '订单编号', '客户', '商品名称', '规格', '数量', '单价', '金额', '含税金额', '订单日期'])
  
  let rowIndex = 1
  orders.forEach(order => {
    // 判断是新订单还是旧订单
    if (order.order_goods && order.order_goods.length > 0) {
      // 新订单：每个商品明细占一行
      order.order_goods.forEach(item => {
        exportData.push([
          rowIndex++,
          order.order_number || '-',
          order.order_client || '-',
          item.goods_name || '-',
          item.spec || '-',
          item.quantity || 0,
          item.price || 0,
          item.amount || 0,
          item.total_amount || 0,
          order.order_date || order.date || '-'
        ])
      })
    } else {
      // 旧订单：一行显示
      exportData.push([
        rowIndex++,
        '-',
        order.order_client || '-',
        order.goods_name || '-',
        '-',
        order.goods_weight || '-',
        '-',
        '-',
        '-',
        order.date || '-'
      ])
    }
  })
  
  // 使用 XLSX 导出
  const ws = XLSX.utils.aoa_to_sheet(exportData)
  const wb = XLSX.utils.book_new()
  XLSX.utils.book_append_sheet(wb, ws, '订单列表')
  XLSX.writeFile(wb, `订单导出_${new Date().toISOString().slice(0, 10)}.xlsx`)
}
```
