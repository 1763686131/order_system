#### 1.2 编辑模式的 OrderForm.vue
需要支持两种模式：
- **新增模式**（现有）：空表单，formData 全部为空

#### 1.3 保存逻辑修改
编辑模式下，保存时需要使用 `PUT` 请求而不是 `POST`：

```javascript
const handleSave = async (printAfterSave = false) => {
  // 1. 校验表单
  if (!validateForm()) return
  
  // 2. 构建请求数据
  const requestData = {
    storeId: formData.value.storeId,
    customerId: formData.value.customerId,
    // ... 其他字段
  }
  
  try {
    let response
    if (isEditMode.value) {
      // 编辑模式：PUT 请求
      response = await request({
        url: `/orders/${props.orderId}`,
        method: 'PUT',
        data: requestData
      })
    } else {
      // 新增模式：POST 请求
      response = await request({
        url: '/orders',
        method: 'POST',
        data: requestData
      })
    }
    
    if (response.success) {
      alert(isEditMode.value ? '订单修改成功！' : '订单保存成功！')
      if (printAfterSave) {
        // TODO: 打印逻辑
      }
      handleClose()
    }
  } catch (error) {
    alert('保存失败：' + (error.message || '未知错误'))
  }
}
```

// 获取商品信息文本
const getGoodsDisplayText = (order) => {
  if (!order) return ''
  
  // 新订单：显示商品明细列表
  if (order.order_goods && order.order_goods.length > 0) {
    const lines = order.order_goods.map((item, index) => 
      `${index + 1}. ${item.goods_name} ${item.spec} x${item.quantity} (${item.packages}件)`
    )
    const totalQty = order.order_goods.reduce((sum, item) => sum + (item.quantity || 0), 0)

  }
  
  // 旧订单：显示原始文本
  return `商品：${order.goods_name || '-'}\n重量：${order.goods_weight || '-'}\n件数：${order.goods_quantity || '-'}`

// 获取运费金额
const getFreightAmountForRow = (order) => {
  // 新订单：从 freight_costs 数组读取
  if (order.freight_costs && Array.isArray(order.freight_costs)) {
    // 如果有多条运费记录，当前行只显示其中一条
    if (order.freightCostIndex !== undefined && order.freight_costs[order.freightCostIndex]) {
      const cost = order.freight_costs[order.freightCostIndex]
      return cost.amount ? `¥${cost.amount.toFixed(2)}` : '-'
    }
    // 如果没有 freightCostIndex，显示第一条
    if (order.freight_costs.length > 0) {
      return order.freight_costs[0].amount ? `¥${order.freight_costs[0].amount.toFixed(2)}` : '-'
    }
  }
  
  // 旧订单：尝试从其他字段读取（如果有的话）
  // 或者返回 '-'，表示需要手动录入
  return '-'
}

// 获取运费备注信息（车牌号、货拉拉等）
const getFreightNote = (order) => {
  if (order.freight_costs && Array.isArray(order.freight_costs)) {
    if (order.freightCostIndex !== undefined && order.freight_costs[order.freightCostIndex]) {
      const cost = order.freight_costs[order.freightCostIndex]
      return cost.note || '-'
    }
    if (order.freight_costs.length > 0) {
      return order.freight_costs[0].note || '-'
    }
  }
  return '-'
}

#### 8.1 订单状态流转
明确订单的状态流转规则：
```
completed（已完成，待出库）
  ↓ 出库
shipped（已出库，待审核）
  ↓ 审核通过
shipped + audit_state=1（已审核，待收款）
  ↓ 全额收款
closed（已关闭/已完成）
```

#### 8.2 权限控制
根据用户角色限制操作权限：
- 普通员工：只能查看，不能编辑
- 财务人员：可以编辑收款信息
- 管理员：可以编辑所有信息

const canEdit = computed(() => {

记录关键操作的日志：
- 订单创建时间和创建人
- 订单编辑时间和编辑人
- 出库时间和操作人

#### 8.4 数据校验增强
在保存和编辑时增强校验：
- 应收金额不能为负数
- 本次收款不能大于应收金额
- 商品数量不能为0
- 单价必须大于0
- 库存不足时给出警告（但允许保存）

```javascript
const validateOrderData = (formData) => {
  const errors = []
  
  // 校验应收金额
  if (formData.shouldReceive < 0) {
    errors.push('应收金额不能为负数')
  }
  
  // 校验本次收款
  if (formData.currentPayment > formData.shouldReceive) {
    errors.push('本次收款不能大于应收金额')
  }
  
  // 校验商品明细
  formData.items.forEach((item, index) => {
    if (item.quantity <= 0) {
      errors.push(`第${index + 1}行商品数量必须大于0`)
    }
    if (item.price <= 0) {
      errors.push(`第${index + 1}行商品单价必须大于0`)
    }
    // 库存不足时警告（不阻止保存）
    if (item.quantity > item.currentStock) {
      console.warn(`第${index + 1}行商品库存不足，当前库存：${item.currentStock}，需要：${item.quantity}`)
    }
  })
  
  return errors
}
```

## 关键注意事项

### ⚠️ 数据一致性
- 编辑订单时必须同时更新新旧字段
- 库存变动要实时反映到库存表
- 财务数据变动要同步更新客户欠款

### ⚠️ 事务处理
- 订单保存、库存扣减、财务记录必须在一个事务中
- 失败时要回滚所有操作
- 后端需要实现事务机制

### ⚠️ 并发控制
- 多人同时编辑同一订单要加锁
- 使用乐观锁（版本号）防止数据覆盖
- 保存时检查数据是否被他人修改

### ⚠️ 性能优化
- 大量订单时使用分页加载
- 对账页面数据量大时使用虚拟滚动
- 财务统计使用缓存，避免重复计算
